cmake_minimum_required(VERSION 3.20)
project(meshcloud VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(meshcloud_core STATIC src/core.cpp src/cid.cpp)
target_link_libraries(meshcloud_core PUBLIC OpenSSL::Crypto)
target_include_directories(meshcloud_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
if(MSVC)
 target_compile_options(meshcloud_core PRIVATE /W4 /WX)
else()
 target_compile_options(meshcloud_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(meshcloud src/main.cpp)
target_link_libraries(meshcloud PRIVATE meshcloud_core)
enable_testing()
add_executable(meshcloud_tests tests/test_core.cpp)
target_link_libraries(meshcloud_tests PRIVATE meshcloud_core)
add_test(NAME security_policy_and_cid COMMAND meshcloud_tests)
