cmake_minimum_required(VERSION 3.20)
project(meshcloud VERSION 0.12.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenSSL REQUIRED COMPONENTS Crypto SSL)
find_package(Threads REQUIRED)
find_package(SQLite3 REQUIRED)
add_library(meshcloud_core STATIC
  src/core.cpp src/cid.cpp src/stream.cpp src/catalog.cpp src/vault.cpp
  src/erasure.cpp src/admission.cpp src/wire.cpp src/exchange.cpp src/published.cpp
  src/p2p_tls.cpp src/p2p_config.cpp src/peer_discovery.cpp src/storage_audit.cpp
  src/audit_net.cpp src/ticket_issue.cpp src/provision.cpp)
target_include_directories(meshcloud_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(meshcloud_core PUBLIC OpenSSL::Crypto OpenSSL::SSL SQLite::SQLite3 Threads::Threads)
if(WIN32)
  target_link_libraries(meshcloud_core PUBLIC ws2_32)
endif()
if(MSVC)
  target_compile_options(meshcloud_core PRIVATE /W4 /WX)
else()
  target_compile_options(meshcloud_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(meshcloud src/main.cpp)
target_link_libraries(meshcloud PRIVATE meshcloud_core)
add_executable(meshcloud_peer src/peer_cli.cpp)
set_target_properties(meshcloud_peer PROPERTIES OUTPUT_NAME "meshcloud-peer")
target_link_libraries(meshcloud_peer PRIVATE meshcloud_core)
add_executable(meshcloud_audit src/auditor_cli.cpp)
set_target_properties(meshcloud_audit PROPERTIES OUTPUT_NAME "meshcloud-audit")
target_link_libraries(meshcloud_audit PRIVATE meshcloud_core)
add_executable(meshcloud_audit_net src/audit_net_cli.cpp)
set_target_properties(meshcloud_audit_net PROPERTIES OUTPUT_NAME "meshcloud-audit-net")
target_link_libraries(meshcloud_audit_net PRIVATE meshcloud_core)
add_executable(meshcloud_ticket src/ticket_cli.cpp)
set_target_properties(meshcloud_ticket PROPERTIES OUTPUT_NAME "meshcloud-ticket")
target_link_libraries(meshcloud_ticket PRIVATE meshcloud_core)
add_executable(meshcloud_provision src/provision_cli.cpp)
set_target_properties(meshcloud_provision PROPERTIES OUTPUT_NAME "meshcloud-provision")
target_link_libraries(meshcloud_provision PRIVATE meshcloud_core)
if(WIN32)
  add_executable(meshcloud_gui WIN32 src/win32_gui.cpp)
  set_target_properties(meshcloud_gui PROPERTIES OUTPUT_NAME "MeshCloud-GUI")
  target_compile_definitions(meshcloud_gui PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(meshcloud_gui PRIVATE meshcloud_core comctl32 comdlg32 shell32 ole32 gdi32 user32)
  if(MSVC)
    target_compile_options(meshcloud_gui PRIVATE /W4 /WX /utf-8)
  endif()
endif()
enable_testing()
add_executable(meshcloud_tests tests/test_core.cpp)
target_link_libraries(meshcloud_tests PRIVATE meshcloud_core)
add_test(NAME security_policy_and_cid COMMAND meshcloud_tests)
add_executable(meshcloud_vault_tests tests/test_vault.cpp)
target_link_libraries(meshcloud_vault_tests PRIVATE meshcloud_core)
add_test(NAME streaming_encrypted_vault_and_retention COMMAND meshcloud_vault_tests)

add_executable(meshcloud_erasure_tests tests/test_erasure.cpp)

add_test(NAME pinned_tls_blockwise_13gib_provisioning_and_guest_refusal COMMAND meshcloud_provision_tests)
set_tests_properties(pinned_tls_blockwise_13gib_provisioning_and_guest_refusal PROPERTIES TIMEOUT 70)
